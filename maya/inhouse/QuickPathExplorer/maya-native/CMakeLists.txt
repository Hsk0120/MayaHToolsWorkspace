cmake_minimum_required(VERSION 3.24)
project(quickPathExplorerNativeMigration LANGUAGES C CXX RC)
option(QPE_PRODUCT_BUILD "Build the production plugin name and command" OFF)
if(QPE_PRODUCT_BUILD)
    add_compile_definitions(QPE_PRODUCT_BUILD)
    set(QPE_TARGET quickPathExplorer)
else()
    set(QPE_TARGET quickPathExplorerNativeMigration)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(/utf-8 /Zc:__cplusplus /permissive-)
include("$ENV{DEVKIT_LOCATION}/cmake/pluginEntry.cmake")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(qpe_zip STATIC "${ROOT}/third_party/miniz/miniz.c" "${ROOT}/third_party/miniz/miniz_tdef.c" "${ROOT}/third_party/miniz/miniz_tinfl.c" "${ROOT}/third_party/miniz/miniz_zip.c")
target_compile_definitions(qpe_zip PUBLIC MINIZ_NO_STDIO MINIZ_NO_ZLIB_COMPATIBLE_NAMES)

if(MAYA_VERSION VERSION_GREATER_EQUAL 2025)
    set(QT_ROOT "$ENV{DEVKIT_LOCATION}/Qt")
    set(QT_MAJOR 6)
else()
    set(QT_ROOT "$ENV{DEVKIT_LOCATION}")
    set(QT_MAJOR 5)
endif()
foreach(component Core Gui Widgets)
    list(APPEND PACKAGE_LIBS "${QT_ROOT}/lib/Qt${QT_MAJOR}${component}.lib")
    list(APPEND INCLUDE_DIRS "${QT_ROOT}/include/Qt${component}")
endforeach()
list(APPEND INCLUDE_DIRS "${QT_ROOT}/include" "$ENV{DEVKIT_LOCATION}/include")
add_compile_definitions(QT_CORE_LIB QT_GUI_LIB QT_WIDGETS_LIB)
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}")
set(moc_includes)
foreach(path IN LISTS INCLUDE_DIRS)
    list(APPEND moc_includes "-I${path}")
endforeach()
set(moc_outputs)
foreach(header core/directory_store core/file_operations browser/browser_window browser/inspector)
    get_filename_component(name "${header}" NAME)
    set(output "${generated}/moc_${name}.cpp")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${QT_ROOT}/bin/moc.exe" ${moc_includes} "${ROOT}/src/${header}.h" -o "${output}"
        DEPENDS "${ROOT}/src/${header}.h")
    list(APPEND moc_outputs "${output}")
endforeach()
foreach(source core/directory_store core/file_operations browser/browser_window)
    get_filename_component(name "${source}" NAME)
    set(output "${generated}/${name}.moc")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${QT_ROOT}/bin/moc.exe" ${moc_includes} "${ROOT}/src/${source}.cpp" -o "${output}"
        DEPENDS "${ROOT}/src/${source}.cpp")
    list(APPEND moc_outputs "${output}")
endforeach()
file(GLOB icons CONFIGURE_DEPENDS "${ROOT}/icons/*")
set(qrc "<RCC><qresource prefix=\"/qpe/icons\">")
foreach(icon IN LISTS icons)
    get_filename_component(name "${icon}" NAME)
    string(APPEND qrc "<file alias=\"${name}\">${icon}</file>")
endforeach()
string(APPEND qrc "</qresource><qresource prefix=\"/qpe/licenses\"><file alias=\"NOTICES.txt\">${ROOT}/third_party/NOTICES.txt</file></qresource></RCC>")
file(GENERATE OUTPUT "${generated}/icons.qrc" CONTENT "${qrc}")
add_custom_command(OUTPUT "${generated}/icons.cpp"
    COMMAND "${QT_ROOT}/bin/rcc.exe" -name qpe_native_icons -o "${generated}/icons.cpp" "${generated}/icons.qrc"
    DEPENDS ${icons} "${ROOT}/third_party/NOTICES.txt" "${generated}/icons.qrc")
file(GLOB sources CONFIGURE_DEPENDS "${ROOT}/src/core/*.cpp" "${ROOT}/src/browser/*.cpp" "${ROOT}/src/maya/*.cpp")
set(SOURCE_FILES ${sources} "${ROOT}/src/native_browser.cpp" ${moc_outputs} "${generated}/icons.cpp")
set(PROJECT_NAME ${QPE_TARGET})
list(APPEND INCLUDE_DIRS "${generated}")
set(LIBRARIES OpenMaya OpenMayaUI Foundation)
build_plugin()
set_target_properties(${QPE_TARGET} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_link_libraries(${QPE_TARGET} qpe_zip ole32 shell32 user32 gdi32 uuid)
# 機能互換の受入完了までは現行製品mllを上書きしない。
if(QPE_PRODUCT_BUILD)
    set(QPE_OUTPUT_ROOT "${ROOT}/.maya-output/product-plugin" CACHE PATH "Staging directory for verified product binaries")
else()
    set(QPE_OUTPUT_ROOT "${ROOT}/.maya-output/native-plugin" CACHE PATH "Fixed validation plugin directory")
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(${QPE_TARGET} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${config} "${QPE_OUTPUT_ROOT}/${MAYA_VERSION}"
        PDB_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_BINARY_DIR}/symbols")
endforeach()

set(QPE_MEDIA_EXECUTABLE "${ROOT}/.maya-output/native-media/qpe-media.exe" CACHE FILEPATH "LGPL media helper built from the supplied sources")
if(NOT EXISTS "${QPE_MEDIA_EXECUTABLE}")
    message(FATAL_ERROR "Build the native media helper with tools/build_native_media.py first")
endif()
configure_file("${ROOT}/native/media.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/media.rc" @ONLY)
target_sources(${QPE_TARGET} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/media.rc")
