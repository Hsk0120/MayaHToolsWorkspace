cmake_minimum_required(VERSION 3.20)
project(QuickPathExplorer LANGUAGES CXX)
if(DEFINED MAYA_VERSION AND NOT MAYA_VERSION STREQUAL "2027")
  message(FATAL_ERROR "This comparison prototype targets Maya 2027 / Qt 6.8.3 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(QPE_QT_ROOT "$ENV{DEVKIT_LOCATION}/Qt" CACHE PATH "Mayaと同じQt SDK")
add_compile_options(/utf-8 /Zc:__cplusplus /permissive- /W4)
add_library(qpeNative SHARED src/explorer.cpp src/benchmark.cpp)
target_compile_definitions(qpeNative PRIVATE QPE_BUILD QT_CORE_LIB QT_GUI_LIB QT_WIDGETS_LIB)
target_include_directories(qpeNative PUBLIC src "${QPE_QT_ROOT}/include")
foreach(component Core Gui Widgets)
  target_include_directories(qpeNative PUBLIC "${QPE_QT_ROOT}/include/Qt${component}")
  target_link_libraries(qpeNative PUBLIC "${QPE_QT_ROOT}/lib/Qt6${component}.lib")
endforeach()
add_executable(QuickPathExplorer src/main.cpp)
target_link_libraries(QuickPathExplorer PRIVATE qpeNative)
