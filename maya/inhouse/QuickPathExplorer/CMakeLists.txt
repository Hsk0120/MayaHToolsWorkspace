cmake_minimum_required(VERSION 3.20)
project(quickPathExplorer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(/utf-8 /Zc:__cplusplus /permissive-)
include("$ENV{DEVKIT_LOCATION}/cmake/pluginEntry.cmake")
if(MAYA_VERSION VERSION_GREATER_EQUAL 2025)
    foreach(component Core Gui Widgets)
        list(APPEND PACKAGE_LIBS "$ENV{DEVKIT_LOCATION}/Qt/lib/Qt6${component}.lib")
        list(APPEND INCLUDE_DIRS "$ENV{DEVKIT_LOCATION}/Qt/include/Qt${component}")
    endforeach()
    list(APPEND INCLUDE_DIRS "$ENV{DEVKIT_LOCATION}/Qt/include")
    set(QPE_RCC "$ENV{DEVKIT_LOCATION}/Qt/bin/rcc.exe")
    add_compile_definitions(QT_CORE_LIB QT_GUI_LIB QT_WIDGETS_LIB)
else()
    set(CMAKE_MAP_IMPORTED_CONFIG_DEBUG "Release")
    macro(_qt5_Core_check_file_exists file)
    endmacro()
    macro(_qt5_Gui_check_file_exists file)
    endmacro()
    macro(_qt5_Widgets_check_file_exists file)
    endmacro()
    set(Qt5_NO_LINK_QTMAIN ON)
    set(CMAKE_PREFIX_PATH "$ENV{DEVKIT_LOCATION}/lib/cmake/Qt5")
    find_package(Qt5 COMPONENTS Core Gui Widgets REQUIRED)
    list(APPEND PACKAGE_LIBS Qt5::Core Qt5::Gui Qt5::Widgets)
    list(APPEND INCLUDE_DIRS "$ENV{DEVKIT_LOCATION}/include")
    set(QPE_RCC "$ENV{DEVKIT_LOCATION}/bin/rcc.exe")
endif()
file(GLOB QPE_PYTHON CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/python/*.py")
file(GLOB QPE_ICONS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/icons/*")
set(QPE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${QPE_GENERATED}")
add_custom_command(OUTPUT "${QPE_GENERATED}/sources.h"
    COMMAND "${CMAKE_COMMAND}" "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/python"
        "-DOUTPUT=${QPE_GENERATED}/sources.h" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake"
    DEPENDS ${QPE_PYTHON} cmake/embed.cmake)
set(qrc "<RCC><qresource prefix=\"/qpe/icons\">")
foreach(icon IN LISTS QPE_ICONS)
    get_filename_component(name "${icon}" NAME)
    string(APPEND qrc "<file alias=\"${name}\">${icon}</file>")
endforeach()
string(APPEND qrc "</qresource></RCC>")
file(GENERATE OUTPUT "${QPE_GENERATED}/icons.qrc" CONTENT "${qrc}")
add_custom_command(OUTPUT "${QPE_GENERATED}/icons.cpp"
    COMMAND "${QPE_RCC}" -name qpe_icons -o "${QPE_GENERATED}/icons.cpp" "${QPE_GENERATED}/icons.qrc"
    DEPENDS ${QPE_ICONS} "${QPE_GENERATED}/icons.qrc")
add_library(qpe_icons OBJECT "${QPE_GENERATED}/icons.cpp")
target_include_directories(qpe_icons PRIVATE ${INCLUDE_DIRS})
target_link_libraries(qpe_icons PRIVATE ${PACKAGE_LIBS})
set(PROJECT_NAME quickPathExplorer)
set(SOURCE_FILES src/plugin.cpp src/native_browser.cpp src/native_browser.h
    "${QPE_GENERATED}/sources.h" $<TARGET_OBJECTS:qpe_icons>)
list(APPEND INCLUDE_DIRS "${QPE_GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}/src")
set(LIBRARIES OpenMaya OpenMayaUI Foundation)
build_plugin()
target_link_libraries(quickPathExplorer ole32 shell32 user32 gdi32)
add_executable(qpe_native_test tests/native_test.cpp src/native_browser.cpp)
target_include_directories(qpe_native_test PRIVATE ${INCLUDE_DIRS})
target_link_libraries(qpe_native_test PRIVATE ${PACKAGE_LIBS} ole32 shell32 user32 gdi32)
# Mayaを使わないPySide統合テスト専用。配布物には含めない。
add_library(qpe_native_bridge_test SHARED tests/bridge.cpp src/native_browser.cpp $<TARGET_OBJECTS:qpe_icons>)
target_include_directories(qpe_native_bridge_test PRIVATE ${INCLUDE_DIRS})
target_link_libraries(qpe_native_bridge_test PRIVATE ${PACKAGE_LIBS} ole32 shell32 user32 gdi32 dbghelp)
# 旧比較用ランチャーの入口を維持する。これらはmllの配布先へ入れない。
if(MAYA_VERSION STREQUAL "2027")
    add_library(qpeNative SHARED src/explorer.cpp src/benchmark.cpp)
    target_compile_definitions(qpeNative PRIVATE QPE_BUILD)
    target_include_directories(qpeNative PUBLIC src ${INCLUDE_DIRS})
    target_link_libraries(qpeNative PUBLIC ${PACKAGE_LIBS})
    add_executable(qpe_comparison src/main.cpp)
    target_link_libraries(qpe_comparison PRIVATE qpeNative)
    set_target_properties(qpe_comparison PROPERTIES OUTPUT_NAME QuickPathExplorer)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(quickPathExplorer PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/release/plug-ins/windows/${MAYA_VERSION}"
        PDB_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_BINARY_DIR}/pdb")
endforeach()
