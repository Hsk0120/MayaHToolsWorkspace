cmake_minimum_required(VERSION 3.24)
project(QuickPathExplorerNative VERSION 1.0.0 LANGUAGES C CXX RC)
option(QPE_PRODUCT_BUILD "Use the production application identity" OFF)
option(QPE_ENABLE_ASAN "Instrument QPE code for memory lifetime regression (development only)" OFF)
if(QPE_ENABLE_ASAN)
    add_compile_options(/fsanitize=address)
    add_link_options(/INCREMENTAL:NO)
    # 配布用QtはASan非計装のSTLを使う。容量境界の追加注釈だけ揃え、QPEの寿命検査は有効にする。
    add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION)
endif()
if(QPE_PRODUCT_BUILD)
    add_compile_definitions(QPE_PRODUCT_BUILD)
endif()
# VSLANG=1033でも英語MUI未導入のMSVCは日本語を返す。
# 実際の出力をUTF-8で読み、Ninjaがヘッダー依存を落とさないようにする。
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qpe_dependency_probe.cpp" "#include <stddef.h>\nint qpe_dependency_probe() { return 1; }\n")
execute_process(COMMAND "${CMAKE_CXX_COMPILER}" /nologo /c /showIncludes
    "${CMAKE_CURRENT_BINARY_DIR}/qpe_dependency_probe.cpp" "/Fo${CMAKE_CURRENT_BINARY_DIR}/qpe_dependency_probe.obj"
    RESULT_VARIABLE qpe_probe_result OUTPUT_VARIABLE qpe_probe_text ERROR_VARIABLE qpe_probe_error ENCODING UTF-8)
string(REGEX MATCH "([^\r\n]+) +[A-Za-z]:[/\\\\][^\r\n]+" qpe_probe_line "${qpe_probe_text}")
if(NOT qpe_probe_result EQUAL 0 OR NOT qpe_probe_line)
    message(FATAL_ERROR "Cannot detect MSVC header dependency prefix: ${qpe_probe_error}")
endif()
string(STRIP "${CMAKE_MATCH_1}" qpe_probe_prefix)
set(CMAKE_CL_SHOWINCLUDES_PREFIX "${qpe_probe_prefix} ")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Svg Test)
get_target_property(QPE_QT_TYPE Qt6::Core TYPE)
if(NOT QPE_QT_TYPE STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "The native Windows build requires static Qt. Do not use Maya's Qt SDK here.")
endif()
if(NOT EXISTS "${QPE_QT_SOURCE_DIR}/LICENSES/LGPL-3.0-only.txt")
    message(FATAL_ERROR "Set QPE_QT_SOURCE_DIR to the corresponding QtBase source tree.")
endif()
add_compile_options(/utf-8 /Zc:__cplusplus /permissive- /Zi)
add_link_options(/DEBUG)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(qpe_zip STATIC "${ROOT}/third_party/miniz/miniz.c" "${ROOT}/third_party/miniz/miniz_tdef.c" "${ROOT}/third_party/miniz/miniz_tinfl.c" "${ROOT}/third_party/miniz/miniz_zip.c")
target_compile_definitions(qpe_zip PUBLIC MINIZ_NO_STDIO MINIZ_NO_ZLIB_COMPATIBLE_NAMES)

file(GLOB core_sources CONFIGURE_DEPENDS "${ROOT}/src/core/*.cpp" "${ROOT}/src/core/*.h")
add_library(qpe_cpp_core STATIC ${core_sources})
target_link_libraries(qpe_cpp_core PUBLIC qpe_zip Qt6::Core Qt6::Gui ole32 shell32 uuid)
file(GLOB browser_sources CONFIGURE_DEPENDS "${ROOT}/src/browser/*.cpp" "${ROOT}/src/browser/*.h")
add_library(qpe_cpp_browser STATIC ${browser_sources} "${ROOT}/src/native_browser.cpp")
target_link_libraries(qpe_cpp_browser PUBLIC qpe_cpp_core Qt6::Widgets Qt6::Svg ole32 shell32 user32 gdi32)
qt_add_executable(QuickPathExplorerNative WIN32 "${ROOT}/src/standalone/native_main.cpp" app.rc)
if(QPE_PRODUCT_BUILD)
    set_target_properties(QuickPathExplorerNative PROPERTIES OUTPUT_NAME QuickPathExplorer)
endif()
target_link_libraries(QuickPathExplorerNative PRIVATE qpe_cpp_browser dwmapi)
file(GLOB icons CONFIGURE_DEPENDS "${ROOT}/icons/*")
qt_add_resources(QuickPathExplorerNative "qpe_icons" PREFIX "/qpe/icons" BASE "${ROOT}/icons" FILES ${icons})
qt_add_resources(QuickPathExplorerNative "qpe_licenses" PREFIX "/qpe/licenses" BASE "${ROOT}/third_party"
    FILES "${ROOT}/third_party/NOTICES.txt")
enable_testing()
qt_add_executable(qpe_cpp_test "${ROOT}/tests/cpp_migration_test.cpp")
set_target_properties(qpe_cpp_test PROPERTIES WIN32_EXECUTABLE OFF)
target_link_libraries(qpe_cpp_test PRIVATE qpe_cpp_browser Qt6::Test)
qt_add_resources(qpe_cpp_test "qpe_test_icons" PREFIX "/qpe/icons" BASE "${ROOT}/icons" FILES ${icons})
add_test(NAME native_migration COMMAND qpe_cpp_test -o "${CMAKE_CURRENT_BINARY_DIR}/test-results.txt,txt")

set(QPE_MEDIA_EXECUTABLE "${ROOT}/.maya-output/native-media/qpe-media.exe" CACHE FILEPATH "LGPL media helper built from the supplied sources")
if(NOT EXISTS "${QPE_MEDIA_EXECUTABLE}")
    message(FATAL_ERROR "Build the native media helper with tools/build_native_media.py first")
endif()
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/media.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/media.rc" @ONLY)
target_sources(QuickPathExplorerNative PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/media.rc")
target_sources(qpe_cpp_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/media.rc")
